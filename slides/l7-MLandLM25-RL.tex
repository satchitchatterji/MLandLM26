\begin{frame}{Types of learning algorithms}
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{.30\textwidth}
\raggedright
\textbf{Supervised learning}

\vspace{.4em}
\textbf{Signal:} labeled data $(x,y)$

\vspace{.4em}
\textbf{Learn:} a predictive mapping
\[
\widehat{y}=f(x)\approx y
\]

\textbf{Examples:} image recognition, sentiment analysis, spam detection...

\column{.02\textwidth}
\centering
\textcolor{darkgreen}{\rule{0.5pt}{0.60\textheight}}

\column{.30\textwidth}
\raggedright
\textbf{Unsupervised learning}

\vspace{.4em}
\textbf{Signal:} unlabeled data $x$

\vspace{.4em}
\textbf{Learn:} structure or representation
\[
z=g(x)
\]

\textbf{Examples:} document clustering, dimensionality reduction, anomaly detection...

\column{.02\textwidth}
\centering
\textcolor{darkgreen}{\rule{0.5pt}{0.60\textheight}}

\column{.30\textwidth}
\raggedright
\textbf{Reinforcement learning}

\vspace{.4em}
\textbf{Signal:} rewards from interaction

\vspace{.4em}
\textbf{Learn:} values or a policy
\[
\begin{aligned}
Q(s,a)&\approx\text{future reward},\\
\pi(a\mid s)&\text{ chooses actions.}
\end{aligned}
\]

\textbf{Examples:} robot navigation, game playing, dialogue systems...
\end{columns}

\begin{frame}{Q-learning and temporal-difference learning}
How do we learn the utility of state--action pairs by trial and error?

\begin{block}{Learning the Q-value}
\[
Q(S,A) = \widehat{U}(S,A)
\]
\end{block}

Q-learning approximates utility through online learning: the policy is applied and learned at the same time.
\end{frame}

\begin{frame}{Q-learning update rule}
\small
\begin{block}{Prediction error}
\[
\delta = R + \gamma \max_{A'} Q_t(S',A') - Q_t(S,A)
\]
\end{block}
\begin{block}{Q-value update}
\[
Q_{t+1}(S,A) = Q_t(S,A) + \alpha\delta
\]
\end{block}
\begin{block}{Asymptotic relationship}
\[
\lim_{t\to\infty} Q_t(S,A) = U(S,A)
\]
\end{block}
\small
Reminder: $U(S,A)=R(S,A)+\gamma\max_{A'}[U(S',A')]$.

\vspace{-1em}
\missingfigure{State-transition diagram from source slide 13}
\end{frame}

\begin{frame}{}
\begin{block}{Q-learning action-selection policy}
Suppose we are in state $S$, with possible actions $A_1$ and $A_2$, and
\[
Q(S,A_1)=0.5, \qquad Q(S,A_2)=0.7.
\]
Which action should we choose?
\end{block}
\pause
\begin{block}{Question 3}
How do you make sure you maximize reward, both in the short term (taking advantage of your knowledge) and in the long term (extending your knowledge)?
\end{block}
\end{frame}

\begin{frame}{Q-learning action-selection policies}
Suppose $Q(S,A_1)=0.5$ and $Q(S,A_2)=0.7$. How should we trade off exploration and exploitation?

\begin{block}{Epsilon-greedy}
\[
P(a_i\mid S)=
\begin{cases}
1-\epsilon & \text{if }q_i=\max_A q_i(A),\\
\dfrac{\epsilon}{|A|-\max_A q_i(A)} & \text{otherwise.}
\end{cases}
\]
The best action is chosen with probability $1-\epsilon$.
\end{block}

\begin{block}{Softmax}
\[
P(a_i\mid S)=\frac{e^{q_i/\tau}}{\sum_A e^{q_i/\tau}}
\]
For $\tau\to0$ the policy is greedy; for $\tau\to\infty$ it is random.
\end{block}
\end{frame}

\begin{frame}{Q-learning pseudo-code}
\begin{enumerate}
    \item Set $\alpha\in[0,1]$, small $\epsilon>0$, and $\gamma>0$.
    \item Initialize $Q(s,a)$ for all $s\in S$ and $a\in A$ to $0$.
    \item Initialize $s$.
    \item For each episode:
    \begin{enumerate}
        \item While $s$ is not terminal:
        \begin{enumerate}
            \item Choose $a$ from $s$ using the action-selection policy.
            \item Observe reward $R$ and next state $s'$.
            \item Update $Q$ and set $s\leftarrow s'$.
        \end{enumerate}
    \end{enumerate}
\end{enumerate}
\end{frame}

\begin{frame}{Model-based versus model-free learning}
\begin{columns}[T]
\column{.47\linewidth}
\begin{block}{Model-free (habitual)}
Rely on trial and error. Efficient, but inflexible.
\end{block}
\column{.47\linewidth}
\begin{block}{Model-based (goal-directed)}
Build a model of the environment. Effortful, but flexible.
\end{block}
\end{columns}

\vspace{1em}
\begin{center}
\textit{Dayan \& Niv (2008)}
\end{center}
\end{frame}

\begin{frame}{Model-based reinforcement learning}

\end{block}

\begin{frame}{Dyna-Q pseudo-code}
\begin{enumerate}
    \item Initialize parameters.
    \item For each episode, while $s$ is not terminal:
    \begin{enumerate}
        \item Choose $a$ from $s$ using the action-selection policy.
        \item Observe $R$ and $s'$, update $Q$, and set $s\leftarrow s'$.
        \item Update the model: $M(s,a)=(R,s')$.
        \item Repeat $n$ times:
        \begin{enumerate}
            \item Choose $(s,a)$ to simulate.
            \item Simulate the action and observe $R$ and $s'$.
            \item Update $Q$.
        \end{enumerate}
    \end{enumerate}
\end{enumerate}
\vspace{.3em}
\small Direct RL update and model learning use real experience; planning uses simulated experience.
\end{frame}

\begin{frame}{Extensions of RL algorithms}
\begin{itemize}
    \item \textbf{Eligibility traces}: update not only the last state--action pair, but all previously visited pairs with a decay factor $\lambda$.
    \item \textbf{Decaying learning rate}: $\alpha$ decreases over time.
    \item \textbf{Hybrid models}: combine model-based and model-free components.
\end{itemize}

\begin{frame}{Deep Q-Learning (Mnih et al. 2015)}
\begin{itemize}
    \item Replace Q-table with a  neural network $Q$, with weights $\theta$; use it to compute Q-value of current state
    \item Make a copy of the Q-network (the \emph{target-network} $\hat{Q}$); use it to compute Q-value of target states
    \item Learn in batches: act in the world (choose actions using $\epsilon$-greedy) and gather data 
    \item Optimize Q-network (mimize prediction error) via gradient descent on a batch of observations
    \item Occasionally copy $Q$ into $\hat{Q}$
\end{itemize}
\end{frame}
