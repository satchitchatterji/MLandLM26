\begin{frame}{Types of Learning Algorithms}
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{.30\textwidth}
\raggedright
\textbf{Supervised learning}

\vspace{.4em}
\textbf{Signal:} labeled data $(x,y)$

\vspace{.4em}
\textbf{Learn:} a predictive mapping
\[
\widehat{y}=f(x)\approx y
\]

\textbf{Examples:} image recognition, sentiment analysis, spam detection...

\column{.02\textwidth}
\centering
\textcolor{darkgreen}{\rule{0.5pt}{0.60\textheight}}

\column{.30\textwidth}
\raggedright
\textbf{Unsupervised learning}

\vspace{.4em}
\textbf{Signal:} unlabeled data $x$

\vspace{.4em}
\textbf{Learn:} structure or representation
\[
z=g(x)
\]

\textbf{Examples:} document clustering, dimensionality reduction, anomaly detection...

\column{.02\textwidth}
\centering
\textcolor{darkgreen}{\rule{0.5pt}{0.60\textheight}}

\column{.30\textwidth}
\raggedright
\textbf{Reinforcement learning}

\vspace{.4em}
\textbf{Signal:} rewards from interaction

\vspace{.4em}
\textbf{Learn:} values or a policy
\[
\begin{aligned}
Q(s,a)&\approx\text{future reward},\\
\pi(a\mid s)&\text{ chooses actions.}
\end{aligned}
\]

\textbf{Examples:} robot navigation, game playing, dialogue systems...
\end{columns}

\begin{frame}{Agent-Environment Interaction Loop}
\centering
\includegraphics[width=0.8\linewidth,keepaspectratio]{../figures/agent_environment_interaction.png}
\begin{enumerate}
    \item The agent $p$ observes the current state $s_t$ of the environment.
    \item It chooses an action $a_t$ to perform in that state, according to its policy $\pi(a\mid s)$.
    \item The environment responds with a new state $s_{t+1}$ and a reward $r_{t+1}$.
    \item The agent updates its knowledge based on the new state and reward.
\end{enumerate}

\end{frame}

\section{Markov Decision Processes}
\begin{frame}{How Do We Define an RL Problem? \textit{Markov Decision Processes}}
\footnotesize
\begin{block}{Definition}
A Markov decision process (MDP) is a mathematical model of sequential decision-making:
\[
\mathcal{M}=\bigl(\mathcal{S},\mathcal{A},T,R,\gamma\bigr).
\]
\end{block}
\begin{description}
    \item[$\mathcal{S}$] states
    \item[$\mathcal{A}$] actions
    % \item[$T(s'\mid s,a)$] probability of reaching $s'$ after taking $a$ in $s$
    \item [$T(s,a)$] transition function: returns next state after taking $a$ in $s$
    \item[$R(s)$] immediate reward at state $s$
    \item[$\gamma$] `discount factor' for future rewards
\end{description}

\begin{alertblock}{\normalsize{\textit{Assumption}}}
\textit{Markov property}: the next state $s'$ depends only on the current state $s$ and action $a$.
\end{alertblock}
\end{frame}

\begin{frame}{From State Values to Action Values}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.63\textwidth}
\footnotesize
\textbf{State value:} How good is it to be in state $s$?
\[
U^*(s)=\max_{a\in\mathcal{A}}Q^*(s,a)
\]

\textbf{Action value:} How good is it to take action $a$ in state $s$?
\[
Q^*(s,a)=R(s)+\gamma U^*(T(s,a))
\]

\begin{alertblock}{Why learn $Q$?}
$Q(s,a)$ lets the agent compare actions directly and choose
\[
\pi^*(s)=\arg\max_a Q^*(s,a).
\]
\end{alertblock}

\scriptsize Q-learning learns these action values directly from experience, without first knowing the transition function $T$.
\end{column}

\begin{frame}{Q-Learning Update Rule (\textit{Temporal-Difference Learning})}
\footnotesize
The Q-learning update rule is a form of \textit{temporal-difference learning}, which updates the action value $Q(s,a)$ based on the observed reward and the estimated future rewards. \textit{\tiny{(Watkins \& Dayan, 1992)}}
\begin{block}{Q-value update}
\[
Q_{t+1}(s_t,a_t) = Q_t(s_t,a_t) + \overset{\substack{\text{\tiny learning rate}\\[-0.4ex]\downarrow}}{\alpha}\delta_t
\]
\end{block}
\begin{block}{Prediction (TD) error}
\[
\delta_t = \overbrace{R(s_{t+1}) + \gamma \max_{a\in\mathcal{A}} Q_t(s_{t+1},a)}^{\text{\scriptsize target}}
           \phantom{..} - \overbrace{Q_t(s_t,a_t)}^{\text{\scriptsize current estimate}}
\]
\end{block}
\begin{block}{Convergence in the limit}
\[
\lim_{t\to\infty} Q_t(s,a) = Q^*(s,a)
\]
\end{block}
\footnotesize
For a deterministic transition function: $Q^*(s,a)=R(s)+\gamma\max_{a'\in\mathcal{A}}Q^*(T(s,a),a')$.

\begin{frame}{Q-Learning Exploration Policies}
Suppose $Q(s,a_1)=0.5$ and $Q(s,a_2)=0.7$. How should we trade off exploration and exploitation?

\begin{block}{Epsilon-greedy}
\[
\mathbb{P}(a_i\mid s)=
\begin{cases}
1-\epsilon & \text{, if } a_i=\arg\max_{a'\in\mathcal{A}} Q(s,a'),\\
\dfrac{\epsilon}{|\mathcal{A}|-1} & \text{, otherwise.}
\end{cases}
\]
With probability $1-\epsilon$ choose the greedy action; with probability $\epsilon$ choose uniformly from the remaining. For $\epsilon\to0$ the policy is greedy; for $\epsilon\to1$ it is uniform.
\end{block}

\begin{block}{Softmax/Boltzmann}
\[
\mathbb{P}(a_i\mid s)=\frac{e^{Q(s,a_i)/\tau}}{\sum_{a'\in\mathcal{A}} e^{Q(s,a')/\tau}}
\]
Exponentially scale the action values and normalize to get a probability distribution. For $\tau\to0$ the policy is greedy; for $\tau\to\infty$ it is uniform.
\end{block}
\end{frame}

\begin{frame}{Q-Learning Pseudo-Code}
\begin{enumerate}
    \item Set $\alpha\in(0,1]$, $\gamma\in[0,1)$, and small $\epsilon\in(0,1]$ or $\tau>0$.
    \item Initialize $Q(s,a)=0$ for all $s\in \mathcal{S}$ and $a\in \mathcal{A}$.
    \item Initialize $s$ to a starting state.
    \item For each episode:
    \begin{enumerate}
        \item While $s$ is not terminal:
        \begin{enumerate}
            \item Choose $a$ from $s$ using an exploration policy.
            \item Observe reward $R$ and next state $s'$.
            \item Update $Q$ and set $s\leftarrow s'$.
            \item Update $\alpha$ and $\epsilon$ or $\tau$ if desired.
        \end{enumerate}
    \end{enumerate}
\end{enumerate}
\end{frame}

\begin{frame}{Extensions of Base RL Algorithms}
\small
\begin{itemize}
    \item \textbf{Decaying learning rate}: $\alpha$ decreases over time.
    \item \textbf{Balancing exploration}: $\epsilon$ or $\tau$ decreases over time.
    \item \textbf{$n$-step returns}: use several future rewards before bootstrapping.
    \item \textbf{Eligibility traces}: update not only the last state--action pair, but all previously visited pairs with a decay factor $\lambda$.
    \item \textbf{Experience replay}: store transitions and learn from them more than once.
    \item \textbf{Function approximation}: represent $Q(s,a)$ with a parameterized function $Q_\theta(s,a)$.
    \item \textbf{Double Q-learning}: reduce overestimation from the maximization step.
    \item \textbf{Hybrid models}: combine model-based and model-free components.
    \item \textbf{Policy gradients}: learn a policy $\pi_\theta(a\mid s)$ directly, rather than learning $Q(s,a)$.
    \item \textbf{Deep RL}: use deep neural networks to represent $Q(s,a)$ or $\pi(a\mid s)$ to handle high-dimensional state spaces.
\end{itemize}
\end{frame}

\begin{frame}{Deep Reinforcement Learning (DQN, Mnih et al. 2015)}
\begin{itemize}
    \item \textbf{Training:} Remember that the Q-learning update rule is:
    \[
    Q_{t+1}(s_t,a_t) = Q_t(s_t,a_t) + \alpha\left(R(s_{t+1}) + \gamma \max_{a\in\mathcal{A}} Q_t(s_{t+1},a) - Q_t(s_t,a_t)\right)
    \]
    with the prediction error being the difference between the target and the current estimate:
\[
\delta_t = \overbrace{R(s_{t+1}) + \gamma \max_{a\in\mathcal{A}} Q_t(s_{t+1},a)}^{\text{\scriptsize target}}
           \phantom{..} - \overbrace{Q_t(s_t,a_t)}^{\text{\scriptsize current estimate}}
\]
    \item In DQN, we replace the Q-table with a neural network $Q_\theta(s,a)$ and minimize the prediction error using a separate target network $Q_{\bar\theta}$:
    \[
    \mathcal{L}(\theta) = \mathbb{E}_{(s_t,a_t,r_t,s_{t+1})\sim D}\left[\left(R(s_{t+1}) + \gamma \max_{a\in\mathcal{A}} Q_{\bar\theta}(s_{t+1},a) - Q_\theta(s_t,a_t)\right)^2\right]
    \]
\end{itemize}
\end{frame}

\begin{frame}{Deep Reinforcement Learning (DQN, Mnih et al. 2015)}
\begin{itemize}
    \item \textbf{Experience replay:} act in the world (choose actions using $\epsilon$-greedy) and gather data. Save transitions $(s,a,r,s')$ in a replay buffer.
    Then uniformly sample mini-batches of transitions from the buffer to train the network. This breaks correlations between sequential data and stabilizes training.
    \item Training is done by stochastic gradient descent on the loss function $\mathcal{L}(\theta)$, using the sampled mini-batches:
    \[
    \nabla \mathcal{L}(\theta) \approx \frac{1}{N}\sum_{i=1}^{N}\nabla \left(R(s_{t+1}) + \gamma \max_{a\in\mathcal{A}} Q_{\bar\theta}(s_{t+1},a) - Q_\theta(s_t,a_t)\right)^2
    \]
\end{itemize}
\end{frame}

\begin{frame}{DQN Stabilization: Target Networks}
\small
\begin{block}{Two networks}
The \textbf{online network} $Q_\theta$ learns from gradient updates. The \textbf{target network} $Q_{\bar\theta}$ provides a temporarily fixed target.
\end{block}
\[
y_t = R(s_{t+1}) + \gamma(1-d_t)\max_{a\in\mathcal{A}}Q_{\bar\theta}(s_{t+1},a)
\]
\begin{block}{Slowly update the target}
Every $K$ gradient steps, copy the online weights: \(\bar\theta \leftarrow \theta\). Here $d_t=1$ when the next state is terminal.
\end{block}
\begin{alertblock}{Why?}
Keeping the target fixed for several updates makes learning more stable: the network is not chasing a target that changes at every step.
\end{alertblock}
\end{frame}

\begin{frame}{Reinforcement Learning for Language Models}
\small
\begin{block}{An RL problem over generated text}
\begin{itemize}
    \item \textbf{State:} a prompt and the tokens generated so far.
    \item \textbf{Action:} the next token, sampled from $\pi_\theta(\text{token}\mid\text{context})$.
    \item \textbf{Reward:} a human preference, AI feedback, a verifier, or task success after a completion.
\end{itemize}
\end{block}
\[
\max_\theta\;\mathbb{E}_{y\sim\pi_\theta(\cdot\mid x)}\bigl[r(x,y)\bigr]
\]
\begin{alertblock}{Examples}
RLHF and RLAIF use preference feedback; policy-gradient methods such as PPO or GRPO update the language-model policy.
\end{alertblock}
\end{frame}

\begin{frame}{Frontiers}
\small
\begin{itemize}
    \item \textbf{World models}: learn predictive simulators and plan through imagined futures.
    \item \textbf{Offline RL}: learn from logged data when online trial and error is costly or unsafe.
    \item \textbf{Foundation-model agents}: combine pretrained language and vision models with RL for planning and tool use.
    \item \textbf{Safe RL}: maximize reward while respecting constraints during learning and deployment.
    \item \textbf{Multi-agent RL}: learn to coordinate, compete, and communicate with other agents.
\end{itemize}

\vfill
\begin{alertblock}{Central challenge:}
Build agents that generalize reliably beyond their training environments and improve safely from limited real-world interaction.
\end{alertblock}
\end{frame}
